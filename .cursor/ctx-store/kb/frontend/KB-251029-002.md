## Problem

React components using `useEffect` with object or array dependencies enter infinite re-render loops. This commonly occurs when fetching data based on filter objects or when dependencies are objects created during render.

Example:
```tsx
const OrdersListPage = () => {
  const [orders, setOrders] = useState([]);
  const [filters, setFilters] = useState({ status: 'active', page: 1 });

  useEffect(() => {
    fetchOrders(filters); // This causes infinite loop!
  }, [filters]); // filters is a new object every render
  
  // Component re-renders infinitely
};
```

## Root Cause

In JavaScript, objects and arrays are compared by reference, not by value. Every time a component re-renders:
1. A new `filters` object is created (even with same values)
2. React detects the reference has changed
3. `useEffect` runs again
4. State update triggers re-render
5. Repeat infinitely

## Solution

### Solution 1: Use Primitive Values in Dependency Array (Recommended)
```tsx
const OrdersListPage = () => {
  const [orders, setOrders] = useState([]);
  const [status, setStatus] = useState('active');
  const [page, setPage] = useState(1);

  useEffect(() => {
    fetchOrders({ status, page });
  }, [status, page]); // Primitives are compared by value ✓
};
```

### Solution 2: Use useMemo to Memoize Object
```tsx
const OrdersListPage = () => {
  const [orders, setOrders] = useState([]);
  const [status, setStatus] = useState('active');
  const [page, setPage] = useState(1);

  // Memoize the filters object
  const filters = useMemo(() => ({
    status,
    page,
  }), [status, page]);

  useEffect(() => {
    fetchOrders(filters);
  }, [filters]); // filters only changes when status/page change ✓
};
```

### Solution 3: Use Custom Hook for Data Fetching
```tsx
// hooks/useOrders.ts
export const useOrders = (filters: OrderFilters) => {
  const [orders, setOrders] = useState<Order[]>([]);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  // Memoize the stringified filters to prevent infinite loops
  const filtersKey = useMemo(
    () => JSON.stringify(filters),
    [filters.status, filters.page, filters.limit] // List all filter properties
  );

  useEffect(() => {
    let cancelled = false;

    const loadOrders = async () => {
      setLoading(true);
      setError(null);
      
      try {
        const result = await orderService.getOrders(filters);
        if (!cancelled) {
          setOrders(result.data.orders);
        }
      } catch (err: any) {
        if (!cancelled) {
          setError(err.message);
        }
      } finally {
        if (!cancelled) {
          setLoading(false);
        }
      }
    };

    loadOrders();

    return () => {
      cancelled = true; // Cleanup to prevent state updates after unmount
    };
  }, [filtersKey]); // Depend on stringified version

  return { orders, loading, error };
};

// Usage
const OrdersListPage = () => {
  const filters = { status: 'active', page: 1 };
  const { orders, loading, error } = useOrders(filters);
};
```

### Solution 4: Use useCallback for Functions
```tsx
const OrdersListPage = () => {
  const [orders, setOrders] = useState([]);

  // Memoize the fetch function
  const fetchOrders = useCallback(async (filters: OrderFilters) => {
    const result = await orderService.getOrders(filters);
    setOrders(result.data.orders);
  }, []); // Empty deps if function doesn't depend on any state

  useEffect(() => {
    fetchOrders({ status: 'active', page: 1 });
  }, [fetchOrders]); // fetchOrders is stable ✓
};
```

### Solution 5: Use Ref for Object Comparison
```tsx
const OrdersListPage = () => {
  const [orders, setOrders] = useState([]);
  const [filters, setFilters] = useState({ status: 'active', page: 1 });
  const prevFiltersRef = useRef<OrderFilters>();

  useEffect(() => {
    // Only fetch if filters actually changed (deep comparison)
    if (JSON.stringify(prevFiltersRef.current) !== JSON.stringify(filters)) {
      fetchOrders(filters);
      prevFiltersRef.current = filters;
    }
  }, [filters]);
};
```

### Solution 6: Disable ESLint Rule (Use Sparingly)
```tsx
useEffect(() => {
  fetchOrders(filters);
  // eslint-disable-next-line react-hooks/exhaustive-deps
}, []); // Only run once on mount - be careful!
```

## Common Scenarios

### Scenario 1: API Call with Query Parameters
```tsx
// ❌ BAD
const [params, setParams] = useState({ search: '', sort: 'date' });
useEffect(() => {
  api.search(params);
}, [params]); // Infinite loop

// ✅ GOOD
const [search, setSearch] = useState('');
const [sort, setSort] = useState('date');
useEffect(() => {
  api.search({ search, sort });
}, [search, sort]);
```

### Scenario 2: Array Dependencies
```tsx
// ❌ BAD
const [selectedIds, setSelectedIds] = useState([1, 2, 3]);
useEffect(() => {
  fetchItemsByIds(selectedIds);
}, [selectedIds]); // Infinite loop if array is recreated

// ✅ GOOD
const selectedIdsString = selectedIds.join(',');
useEffect(() => {
  fetchItemsByIds(selectedIds);
}, [selectedIdsString]); // String comparison

// OR use deep comparison library
import { useDeepCompareEffect } from 'use-deep-compare';
useDeepCompareEffect(() => {
  fetchItemsByIds(selectedIds);
}, [selectedIds]);
```

### Scenario 3: Props Object
```tsx
// ❌ BAD
const OrdersList = ({ filters }) => {
  useEffect(() => {
    fetchOrders(filters);
  }, [filters]); // New object every time parent renders
};

// ✅ GOOD - Destructure in parent
<OrdersList status={filters.status} page={filters.page} />

// Or memoize in parent
const FilteredOrdersList = React.memo(({ filters }) => {
  useEffect(() => {
    fetchOrders(filters);
  }, [filters]);
}, (prevProps, nextProps) => {
  return JSON.stringify(prevProps.filters) === JSON.stringify(nextProps.filters);
});
```

## Best Practices

1. **Prefer primitive dependencies**: Use strings, numbers, booleans
2. **Memoize objects with useMemo**: When objects are necessary
3. **Memoize functions with useCallback**: When passing functions as deps
4. **Extract to custom hooks**: Encapsulate complex logic
5. **Use cleanup functions**: Prevent state updates after unmount
6. **Consider libraries**: `use-deep-compare-effect` for complex objects
7. **Understand the issue**: Don't blindly disable ESLint warnings

### Custom Hook for Debugging
```tsx
const useWhyDidYouUpdate = (name: string, props: any) => {
  const previousProps = useRef<any>();

  useEffect(() => {
    if (previousProps.current) {
      const allKeys = Object.keys({ ...previousProps.current, ...props });
      const changedProps: any = {};
      
      allKeys.forEach(key => {
        if (previousProps.current[key] !== props[key]) {
          changedProps[key] = {
            from: previousProps.current[key],
            to: props[key],
          };
        }
      });

      if (Object.keys(changedProps).length > 0) {
        console.log('[why-did-you-update]', name, changedProps);
      }
    }

    previousProps.current = props;
  });
};

// Usage
const OrdersListPage = ({ filters }) => {
  useWhyDidYouUpdate('OrdersListPage', { filters });
  // ...
};
```
